<script>

let participants=[];

function startDraw(){

    let input=document
        .getElementById("namesInput")
        .value;

    participants=input
        .split(",")
        .map(x=>x.trim());

    participants=shuffleArray(participants);

    renderCards();
}

function renderCards(){

    let container=
        document.getElementById("cards");

    container.innerHTML="";

    participants.forEach((name,index)=>{

        let card=document
            .createElement("div");

        card.className="card";

        card.innerHTML=`
        <div class="inner">
            <div class="front">${index+1}</div>
            <div class="back">${name}</div>
        </div>
        `;

        card.onclick=function(){

            if(!card.classList.contains("flipped")){
                card.classList.add("flipped");
            }

        };

        container.appendChild(card);

    });

}

function shuffleCards(){

    participants=
        shuffleArray(participants);

    let cards=
        document.querySelectorAll(".card");

    cards.forEach(c=>{
        c.classList.add("shuffle");
    });

    setTimeout(()=>{
        renderCards();
    },1000);

}

function shuffleArray(arr){

    for(let i=arr.length-1;i>0;i--){

        let j=Math.floor(
            Math.random()*(i+1)
        );

        [arr[i],arr[j]]=
        [arr[j],arr[i]];
    }

    return arr;
}
